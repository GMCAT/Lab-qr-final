1. แนวคิดระบบ
Flow หลัก

แอดมิน เพิ่มข้อมูลอุปกรณ์ 1 ชิ้น = 1 record
ระบบเจน QR Code อัตโนมัติ ผูกกับ ของอุปกรณ์ → idhttps://your-domain.com/item/UUID
ปริ้น QR ไปติดที่เครื่อง
คนในแล็บสแกน → เปิดเว็บหน้าแรกโชว์ข้อมูลย่อ
กด "เพิ่มเติม" → โชว์ข้อมูลเต็ม + คู่มือ/รูป + ประวัติยืม-คืน
อัปโหลด Excel/PDF เพื่อ import อุปกรณ์ทีละหลายชิ้นได้
ระบบยืม-คืน แยกไฟล์/โมดูล ทำทีหลังได้

function formatDate(dateStr) {
  if (!dateStr) return '-';
  const d = new Date(dateStr);
  return `${d.getDate().toString().padStart(2,'0')}/${(d.getMonth()+1).toString().padStart(2,'0')}/${d.getFullYear()}`;
}

function formatPrice(price) {
  if (!price) return '-';
  return new Intl.NumberFormat('th-TH').format(price) + ' บาท';
}

function downloadQR(id) {

    </div>
  `;
}

function select(label, name, value, options) {
  return `
    <div>
      <label class="block text-sm font-semibold mb-1">${label}</label>
      <select name="${name}" class="w-full px-4 py-2 border rounded-lg">
        ${options.map(o => `<option ${o === value? 'selected' : ''}>${o}</option>`).join('')}
      </select>
    </div>
  `;
}

function textarea(label, name, value) {
  return `
    <div>
      <label class="block text-sm font-semibold mb-1">${label}</label>
      <textarea name="${name}" rows="3" class="w-full px-4 py-2 border rounded-lg">${value || ''}</textarea>
    </div>
  `;
}

function saveItem(id, isNew) {
  const form = document.getElementById('itemForm');
  const formData = new FormData(form);
  const db = getDB();

  const newItem = {
    id,
    asset_code: formData.get('asset_code'),
    name: formData.get('name'),
    model: formData.get('model'),
    serial_no: formData.get('serial_no'),
    brand: formData.get('brand'),
    size: formData.get('size'),
    price: Number(formData.get('price')),
    purchase_date: formData.get('purchase_date'),
    status: formData.get('status'),
    location: formData.get('location'),
    responsible: formData.get('responsible'),
    note: formData.get('note'),
    files: isNew? [] : db.items.find(i => i.id === id).files,
    borrow_logs: isNew? [] : db.items.find(i => i.id === id).borrow_logs
  };

  if (isNew) {
    db.items.push(newItem);
  } else {
    const idx = db.items.findIndex(i => i.id === id);
    db.items[idx] = newItem;
  }

  saveDB(db);
  location.hash = '#/admin';
}

}
